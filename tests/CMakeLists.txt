set(VAD_TEST_DATA ${PROJECT_SOURCE_DIR}/models CACHE PATH "Test data from scripts/prepare-test-data.sh")

function(vad_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE vad vad-common ${ARGN})
    target_compile_definitions(${name} PRIVATE VAD_TEST_DATA="${VAD_TEST_DATA}" VAD_ASSETS="${PROJECT_SOURCE_DIR}/assets")
    if(EMSCRIPTEN)
        target_link_options(${name} PRIVATE ${VAD_NODE_LINK_OPTIONS})
    endif()
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES SKIP_RETURN_CODE 77)
endfunction()

vad_test(test_models)
vad_test(test_segments)
vad_test(test_turn)
vad_test(test_diarization)

endif()
